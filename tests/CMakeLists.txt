find_package(Threads REQUIRED)
add_library(graphite_test_harness STATIC test.c)
target_link_libraries(graphite_test_harness PUBLIC graphite)
target_include_directories(graphite_test_harness PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(NOT MSVC)
    target_compile_options(graphite_test_harness PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

function(graphite_add_suite target suite)
    add_executable(${target} ${ARGN})
    target_include_directories(${target} PRIVATE ${PROJECT_SOURCE_DIR}/src/core)
    target_link_libraries(${target} PRIVATE graphite_test_harness Threads::Threads)
    if(NOT MSVC)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME graphite.${suite} COMMAND ${target})
endfunction()

graphite_add_suite(graphite_resource_tests resources resources.c)
graphite_add_suite(graphite_scene_tests scene scene.c lifecycle.c)
graphite_add_suite(graphite_text_tests text text.c)
