cmake_minimum_required(VERSION 3.20)
set(CONSUMER_LANGUAGE C CACHE STRING "Consumer language: C or CXX")
project(graphite_package_consumer LANGUAGES ${CONSUMER_LANGUAGE})
enable_testing()
set(GRAPHITE_COMPONENT graphite CACHE STRING "Installed component to exercise")
find_package(graphite 0.0.1 CONFIG REQUIRED COMPONENTS ${GRAPHITE_COMPONENT})
if(CONSUMER_LANGUAGE STREQUAL "C")
    set(source main.c)
elseif(CONSUMER_LANGUAGE STREQUAL "CXX")
    set(source main.cpp)
else()
    message(FATAL_ERROR "CONSUMER_LANGUAGE must be C or CXX")
endif()
add_executable(consumer ${source})
target_link_libraries(consumer PRIVATE graphite::${GRAPHITE_COMPONENT})
if(NOT GRAPHITE_COMPONENT STREQUAL "graphite")
    string(TOUPPER ${GRAPHITE_COMPONENT} platform)
    target_compile_definitions(consumer PRIVATE CONSUMER_${platform})
endif()
add_test(NAME consumer COMMAND consumer)
