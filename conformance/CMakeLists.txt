find_package(Threads REQUIRED)
add_library(graphite_conformance STATIC cases.c context.c frame.c semantic.c measured.c text.c)
target_link_libraries(graphite_conformance PUBLIC graphite Threads::Threads)
target_include_directories(graphite_conformance PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(graphite_conformance PRIVATE GRAPHITE_CONFORMANCE_OUT="${PROJECT_SOURCE_DIR}/target/conformance")
if(NOT MSVC)
    target_compile_options(graphite_conformance PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
    add_executable(graphite_conformance_tests tests.c)
    target_link_libraries(graphite_conformance_tests PRIVATE graphite_conformance graphite_test_harness)
    add_test(NAME graphite.conformance.helpers COMMAND graphite_conformance_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME graphite.conformance.compare
            COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/compare_test.py)
    endif()
endif()
if(APPLE)
    add_executable(graphite_macos_conformance macos.m)
    target_link_libraries(graphite_macos_conformance PRIVATE graphite_conformance graphite_macos "-framework AppKit")
    target_compile_options(graphite_macos_conformance PRIVATE -fobjc-arc)
    if(BUILD_TESTING)
        add_test(NAME graphite.conformance.macos COMMAND graphite_macos_conformance)
    endif()
endif()
if(TARGET graphite_gtk)
    add_executable(graphite_gtk_conformance gtk.c)
    target_link_libraries(graphite_gtk_conformance PRIVATE graphite_conformance graphite_gtk)
    target_compile_options(graphite_gtk_conformance PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(BUILD_TESTING)
        add_test(NAME graphite.conformance.gtk COMMAND graphite_gtk_conformance)
    endif()
endif()
if(TARGET graphite_windows)
    add_executable(graphite_windows_conformance windows.c)
    target_link_libraries(graphite_windows_conformance PRIVATE graphite_conformance graphite_windows)
    if(BUILD_TESTING)
        add_test(NAME graphite.conformance.windows COMMAND graphite_windows_conformance)
        set_tests_properties(graphite.conformance.windows PROPERTIES RUN_SERIAL TRUE TIMEOUT 300)
    endif()
endif()
if(TARGET graphite_web)
    # Runs in headless Chromium through tests/web/run.mjs; JSPI lets captures wait for the page's screenshots.
    add_executable(graphite_web_conformance web.c)
    target_link_libraries(graphite_web_conformance PRIVATE graphite_conformance graphite_web)
    target_compile_options(graphite_web_conformance PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(graphite_web_conformance PRIVATE -sJSPI -sSINGLE_FILE=1 -sEXIT_RUNTIME=1 -sENVIRONMENT=web
        -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=1MB "SHELL:--js-library ${CMAKE_CURRENT_SOURCE_DIR}/web.js")
    set_property(TARGET graphite_web_conformance APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/web.js)
    if(BUILD_TESTING AND GRAPHITE_NODE AND EXISTS ${PROJECT_SOURCE_DIR}/tests/web/node_modules/playwright)
        add_test(NAME graphite.conformance.web
            COMMAND ${GRAPHITE_NODE} ${PROJECT_SOURCE_DIR}/tests/web/run.mjs $<TARGET_FILE:graphite_web_conformance>)
    endif()
endif()
