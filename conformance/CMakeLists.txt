add_library(graphite_conformance STATIC cases.c context.c frame.c semantic.c measured.c)
target_link_libraries(graphite_conformance PUBLIC graphite)
target_include_directories(graphite_conformance PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(graphite_conformance PRIVATE GRAPHITE_CONFORMANCE_OUT="${PROJECT_SOURCE_DIR}/target/conformance")
if(NOT MSVC)
    target_compile_options(graphite_conformance PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
    add_executable(graphite_conformance_tests tests.c)
    target_link_libraries(graphite_conformance_tests PRIVATE graphite_conformance graphite_test_harness)
    add_test(NAME graphite.conformance.helpers COMMAND graphite_conformance_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME graphite.conformance.compare
            COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/compare_test.py)
    endif()
endif()
if(APPLE)
    add_executable(graphite_macos_conformance macos.m)
    target_link_libraries(graphite_macos_conformance PRIVATE graphite_conformance graphite_macos "-framework AppKit")
    target_compile_options(graphite_macos_conformance PRIVATE -fobjc-arc)
    if(BUILD_TESTING)
        add_test(NAME graphite.conformance.macos COMMAND graphite_macos_conformance)
    endif()
endif()
