add_library(graphite_example STATIC example.c)
target_include_directories(graphite_example PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(graphite_example PUBLIC graphite)
set_target_properties(graphite_example PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
if(NOT MSVC)
    target_compile_options(graphite_example PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
set_property(GLOBAL APPEND PROPERTY GRAPHITE_EXAMPLE_TARGETS graphite_example)

if(APPLE)
    add_library(graphite_example_shell STATIC cli.c frames.c platform/macos.m)
    target_link_libraries(graphite_example_shell PUBLIC graphite_example graphite_macos "-framework ImageIO")
    target_compile_options(graphite_example_shell PRIVATE -Wall -Wextra -Wpedantic -Werror
        "$<$<COMPILE_LANGUAGE:OBJC>:-fobjc-arc>" "$<$<COMPILE_LANGUAGE:OBJC>:-fblocks>")
elseif(WIN32)
    add_library(graphite_example_shell STATIC cli.c frames.c platform/windows.cpp)
    target_compile_features(graphite_example_shell PRIVATE cxx_std_20)
    target_compile_definitions(graphite_example_shell PRIVATE NOMINMAX UNICODE _UNICODE)
    target_link_libraries(graphite_example_shell PUBLIC graphite_example graphite_windows windowscodecs)
    target_compile_options(graphite_example_shell PRIVATE /W4 /WX /EHsc)
elseif(TARGET graphite_gtk)
    add_library(graphite_example_shell STATIC cli.c frames.c platform/gtk.c)
    target_link_libraries(graphite_example_shell PUBLIC graphite_example graphite_gtk)
    target_compile_options(graphite_example_shell PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(TARGET graphite_example_shell)
    set_target_properties(graphite_example_shell PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    set_property(GLOBAL APPEND PROPERTY GRAPHITE_EXAMPLE_TARGETS graphite_example_shell)
endif()
