find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.22)
add_library(graphite_gtk convert.c draw.c tree.c host.c widget.c offscreen.c text.c)
add_library(graphite::gtk ALIAS graphite_gtk)
target_link_libraries(graphite_gtk PUBLIC graphite PkgConfig::GTK4)
target_compile_features(graphite_gtk PRIVATE c_std_11)
target_compile_definitions(graphite_gtk PRIVATE GRAPHITE_GTK_BUILDING)
target_compile_options(graphite_gtk PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(graphite_gtk PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
if(BUILD_TESTING)
    add_executable(graphite_gtk_tests ${PROJECT_SOURCE_DIR}/tests/gtk.c)
    target_link_libraries(graphite_gtk_tests PRIVATE graphite_gtk graphite_test_harness)
    target_compile_options(graphite_gtk_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME graphite.gtk COMMAND graphite_gtk_tests)
    set_tests_properties(graphite.gtk PROPERTIES RUN_SERIAL TRUE TIMEOUT 30)
endif()
